10    // Given the PLONK Constraints, which have the following form: qL*a + qR*b + qM*a*b + qO*c + qC = 0,
11    // calculate the number of constraints required in the compressed Plonk. 
12    // Since each regular plonk constrain only uses 3 wires (a, b and c), and several sets of wires can share the same set of polynomial
13    // gates, we can further extend the compression by storing 2 different sets of (a_i, b_i, c_i) for every set of (qL, qR, qM, Q0, qC)
14    // the committed polynomial. 
15    // In this particular case, we will store two sets of gates in every row. The first one will correspond to a[0], a[1], a[2] and a[3], a[4], a[5] 
16    // and the second on will correspond to a[6], a[7], a[8] and a[9], a[10], a[11]
17    // In addition to that, since the first row of TreeSelector custom gate only uses 6 committed polynomials and it doesn't use any constant value, we will 
18    // verify two plonk constraints in those rows.
19    // In the same way, since Cmul and EvalPol have 3 committed polynomial values empty we will verify two plonk constraints in each row
20
21    // Get information about the custom gates from the R1CS
22    const customGatesInfo = getCustomGatesInfo(r1cs);
23    
24    let NUsed;
25    
26    // Calculate the total number of publics used in PIL and how many rows are needed to store all of them (remember that each row can store up to 12 values)
27    let nPublics = r1cs.nOutputs + r1cs.nPubInputs;
28    let nPublicRows; 
29
30    let nCMulRows;
31    let nPoseidon12Rows;
32    let nCustPoseidon12Rows;
33    let nTotalPoseidon12Rows;
34    let nFFT4Rows;
35    let nEvPol4Rows;
36    let nTreeSelector4Rows;
37
38    if(cols === 12) {
39        // Each Poseidon12 custom gate uses 11 rows (Input -> Round 1 -> Round 2 -> Round 3 -> Round 4 -> Round 15 -> Round 26 -> Round 27 -> Round 28 -> Round 29 -> Output)
40        nPublicRows = Math.floor((nPublics + 11)/12);
41        nCMulRows = customGatesInfo.nCMul;
42        nPoseidon12Rows = customGatesInfo.nPoseidon12*11;
43        nCustPoseidon12Rows = customGatesInfo.nCustPoseidon12*11;
44        nTotalPoseidon12Rows = nPoseidon12Rows + nCustPoseidon12Rows;
45        nFFT4Rows = customGatesInfo.nFFT4*2; // Each FFT4 custom gate uses 2 rows (1 for actually computing the FFT and the other one for checking the output)
46        nEvPol4Rows = customGatesInfo.nEvPol4*2; // Each EvalPol4 custom gate uses 2 rows (1 for actually computing the evaluation and the other one for checking the output)
47        nTreeSelector4Rows = customGatesInfo.nTreeSelector4*2;  // Each TreeSelector custom gate uses 2 rows
48
49        const nPartialRowsCustomGates = nPublicRows + nCMulRows +nTotalPoseidon12Rows + nFFT4Rows + nEvPol4Rows + nTreeSelector4Rows;
50        
51        // Calculate how many  constraints are needed 
52        const CPlonkConstraints = calculatePlonkConstraintsRowsC12(plonkConstraints, customGatesInfo.nEvPol4 + customGatesInfo.nCMul, customGatesInfo.nTreeSelector4);
53
54        console.log(`Number of Plonk constraints stored in EvalPol -> ${customGatesInfo.nEvPol4*3}`)
55        console.log(`Number of Plonk constraints stored in CMul and FFT4 -> ${(customGatesInfo.nCMul + customGatesInfo.nFFT4)*2}`)   
56        console.log(`Number of plonk constraints new rows: ${CPlonkConstraints}`);
57        
58        NUsed = nPartialRowsCustomGates + CPlonkConstraints;
59    } else if(cols === 21) {
60        // Each Poseidon12 custom gate uses 6 rows (Input -> Round 2 -> Round 4 -> Round 26 -> Round 28 -> Output)
61        nPublicRows = Math.floor((nPublics + 20)/21);
62        nCMulRows = Math.ceil(customGatesInfo.nCMul/2);
63        nPoseidon12Rows = customGatesInfo.nPoseidon12*6;
64        nCustPoseidon12Rows = customGatesInfo.nCustPoseidon12*6;
65        nTotalPoseidon12Rows = nPoseidon12Rows + nCustPoseidon12Rows;
66        nFFT4Rows = customGatesInfo.nFFT4*2;
67        nEvPol4Rows = customGatesInfo.nEvPol4;
68        nTreeSelector4Rows = customGatesInfo.nTreeSelector4;
69        
70        // Calculate how many groups of two plonk constraints can be made 
71        const CPlonkConstraints = calculatePlonkConstraintsRowsC21(plonkConstraints, customGatesInfo.nPoseidon12*6 + customGatesInfo.nCustPoseidon12*5 + nFFT4Rows, customGatesInfo.nCustPoseidon12, nCMulRows + nTreeSelector4Rows);
72
73        console.log(`Number of plonk constraints: ${plonkConstraints.length}`); 
74        console.log(`Number of Plonk constraints stored in rows -> ${CPlonkConstraints}`);
75        console.log(`Number of plonk constraints stored in custom gates: ${plonkConstraints.length - CPlonkConstraints}`);
76
77        NUsed = CPlonkConstraints + nPublicRows + nCMulRows + nTotalPoseidon12Rows + nFFT4Rows + nEvPol4Rows + nTreeSelector4Rows;
78    } else throw new Error("Invalid number of cols");
79

81    console.log(`Number of CMul: ${customGatesInfo.nCMul} -> Constraints: ${nCMulRows}`);
82    console.log(`Number of Poseidon12: ${customGatesInfo.nPoseidon12} -> Constraints: ${nPoseidon12Rows}`);
