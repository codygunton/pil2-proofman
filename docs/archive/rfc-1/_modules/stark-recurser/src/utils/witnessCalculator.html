 70class ColsPil2 {
 71    constructor(symbols, degree) {
 72        this.$$def = {};
 73        this.$$defArray = [];
 74
 75        this.F = new F3g();
 76        this.$$n = degree;
 77        this.$$nCols = symbols.length;
 78        this.$$buffer = [];
 79                    
 80        this.$$buffer = new BigBuffer(this.$$nCols*this.$$n);
 81
 82        this.symbols = symbols;
 83        for(let i = 0; i < symbols.length; ++i) {
 84            const symbol = symbols[i];
 85            const name = symbol.name;
 86            const [nameSpace, namePol] = name.split(".");
 87            if (!this[nameSpace]) this[nameSpace] = {};
 88            if (!this.$$def[nameSpace]) this.$$def[nameSpace] = {};
 89            
 90            const polProxy = this.createArrayProxy(symbol.id);
 91
 92            this.$$defArray[symbol.id] = {
 93            name: name,
 94            id: symbol.id,
 95            stage: symbol.stage,
 96            polDeg: degree,
 97            lengths: symbol.lengths
 98            };
 99
100            if (symbol.lengths.length > 0) {
101                if (!this[nameSpace][namePol]) this[nameSpace][namePol] = [];
102                if (!this.$$def[nameSpace][namePol]) this.$$def[nameSpace][namePol] = [];
103                setValueMultiArray(this[nameSpace][namePol], symbol.lengths, polProxy);
104                setValueMultiArray(this.$$def[nameSpace][namePol], symbol.lengths, this.$$defArray[symbol.id]);
105            } else {
106                this[nameSpace][namePol] = polProxy;
107                this.$$def[nameSpace][namePol] = this.$$defArray[symbol.id];
108            }
109            this[symbol.id] = polProxy;
110        }

113    createArrayProxy(symbolId) {
114          
115        const nCols = this.$$nCols;
116        const buff = this.$$buffer;
117        const N = this.$$n;
118        const F = this.F;
119
120        return new Proxy([], {
121            set(target, prop, value) {
122                const pos = parseInt(prop, 10);
123                const buffIndex = nCols * pos + symbolId;
124                
125                if(value < 0n) value += F.p;
126
127                buff.setElement(buffIndex,value);
128                
129                target[pos] = value; // This adds the value to the array itself
130                
131                return true;
132            },
133
134            get(target, prop) {
135                if (prop === 'length') {
136                    return N; // Return the degree for 'length' property
137                }
138                const pos = parseInt(prop, 10);
139                const buffIndex = nCols * pos + symbolId;
140                return buff.getElement(buffIndex);
141            }
